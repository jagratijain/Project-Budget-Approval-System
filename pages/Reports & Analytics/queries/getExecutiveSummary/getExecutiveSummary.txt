SELECT 
  COALESCE(SUM(budget_amount), 0) as total_requested,
  COALESCE(SUM(CASE WHEN status = 'approved' THEN budget_amount ELSE 0 END), 0) as total_approved,
  COALESCE(AVG(budget_amount), 0) as avg_project_value,
  COALESCE(ROUND(COUNT(CASE WHEN status = 'approved' THEN 1 END) * 100.0 / NULLIF(COUNT(*), 0), 1), 0) as approval_rate
FROM projects
