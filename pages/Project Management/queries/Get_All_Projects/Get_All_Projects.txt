SELECT 
  p.id,
  p.title,
  p.description,
  p.budget_amount,
  p.status,
  p.priority,
  p.created_date,
  d.name as department_name,
  u.name as created_by_name
FROM projects p
LEFT JOIN departments d ON p.department_id = d.id
LEFT JOIN users u ON p.created_by = u.id  
ORDER BY p.created_date DESC
